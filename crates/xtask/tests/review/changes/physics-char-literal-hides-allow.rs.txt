//! A fixture core crate.

/// Half of a number.
pub fn half(x: f64) -> f64 {
    x * 0.5
}

/// A sine, with a house-rule exception after a char literal that holds a
/// double quote (the Reviewer's case on #92).
#[rustfmt::skip]

    let _q = '"'; #[allow(clippy::disallowed_methods)] let y = x.sin();
